conversation.yamlMULTI-DEVICE
devices:
  alice: { platform: ios, appId: com.example.chat }
  bob: { platform: android, appId: com.example.chat }
---
- parallel:
    alice: [launchApp]
    bob: [launchApp]
- alice:
    - tapOn: "Message"
    - inputText: "Hi Bob!"
    - tapOn: "Send"
    - signal: sent
- bob:
    - waitForSignal: sent
    - assertVisible: "Hi Bob!"
❯ tb test conversation.yaml
